    FieldInformation(Magma<?> field, String... examples) {
        this.field = field;
        this.finite = field.isFinite();

        this.binaryOperators = field.getSupportedOperators()
            .stream()

        log.fine(() -> "Created %s, operators: %s, unary: %s examples: %s, elements: %s".formatted(field,

            List.of(unaryOperators),

    public static String[] elements(Magma<?> field) {
        Set<String> elements = new LinkedHashSet<>(field.getConstants().keySet());
        if (field.getCardinality().isCountable() && field instanceof Streamable<?> streamable) {
            try {
                streamable.stream().limit(100).map(Object::toString).forEach(elements::add);
            } catch (NotStreamable ignored) {
            }
        }
        return elements.toArray(new String[0]);
    }

    /**
     * The keys of an on-screen keypad for this structure.
     * <ul>
     *   <li>{@code element}: for finite structures all elements, otherwise digits, constants, imaginary units, the variable of polynomials,
     *   as far as the structure accepts them</li>
     *   <li>{@code binary}: the binary operators</li>
     *   <li>{@code unary}: the unary operators. {@link Key#insert()} is then a template, where {@link #SLOT} is to be replaced with the operand</li>
     * </ul>
     */
    public Key[] keys() {
        if (keys == null) {
            List<Key> result = new ArrayList<>();
            elementKeys(result);
            boolean hasMultiplication = field.getSupportedOperators().stream().anyMatch(op -> op.getSymbol().equals("⋅"));
            for (AlgebraicBinaryOperator op : field.getSupportedOperators()) {
                if (hasMultiplication && op.getSymbol().equals("*")) {
                    // '*' is the generic group operation, which here is the same as '⋅'
                    continue;
                }
                result.add(new Key(op.getSymbol(), op.getSymbol(), "binary", title(op.name())));
            }
            for (AlgebraicUnaryOperator op : field.getSupportedUnaryOperators()) {
                String template = inputTemplate(op);
                if (template != null) {
                    result.add(new Key(op.stringify(SLOT), template, "unary", title(op.name())));
                }
            }
            keys = result.toArray(new Key[0]);
        }
        return keys;
    }
    private Key[] keys;

    private void elementKeys(List<Key> result) {
        if (finite && field instanceof Streamable<?> streamable) {
            streamable.stream().limit(100).map(Object::toString).forEach(e ->
                result.add(new Key(e, AST.asOperand(e), "element", null))
            );
            return;
        }
        for (int i = 0; i <= 9; i++) {
            result.add(new Key(String.valueOf(i), String.valueOf(i), "element", null));
        }
        if (accepts("1.5")) {
            result.add(new Key(".", ".", "element", "decimal point"));
        }
        Set<String> candidates = new LinkedHashSet<>();
        for (String constant : field.getConstants().keySet()) {
            // names like 'unity', 'zero', 'pi' are aliases, and not useful as keys
            if (!constant.matches("[a-z]{2,}")) {
                candidates.add(constant);
            }
        }
        candidates.addAll(List.of("i", "j", "k"));
        for (String candidate : candidates) {
            if (accepts(candidate)) {
                result.add(new Key(candidate, candidate, "element", null));
            }
        }
        if (field instanceof PolynomialRing<?> polynomialRing) {
            result.add(new Key(polynomialRing.getVariable(), "\"" + polynomialRing.getVariable() + "\"", "element", "the variable"));
        }
    }

    /**
     * Whether the given input is a valid expression for this structure (without logging, failure is normal here)
     */
    @SuppressWarnings({"unchecked", "rawtypes"})
    private boolean accepts(String input) {
        try {
            AST.parse(input, (AlgebraicStructure) field).eval();
            return true;
        } catch (RuntimeException e) {
            return false;
        }
    }

/**
 * In templates for unary operators, the place where the operand goes.
 */
public static final String SLOT = "□";

/**
 * A key on the on-screen keypad.
 * @param label  what is shown on the key
 * @param insert what is inserted in the expression (for unary operators a template containing {@link #SLOT})
 * @param type   'element', 'binary' or 'unary'
 * @param title  a description, or {@code null}
 */
public record Key(String label, String insert, String type, String title) {
}

/**
 * How a unary operator can be entered in the infix notation, with {@link #SLOT} for the operand. Or {@code null} if it can't (or there is no point, like for the identity '+').
 */
static String inputTemplate(AlgebraicUnaryOperator op) {
    String symbol = op.getSymbol();
    String x = TextUtils.PLACEHOLDER;
    if (symbol.matches("[a-z][a-z0-9]*")) {
        // function notation, like sqr(x), sin(x), inverse(x)
        return symbol + "(" + SLOT + ")";
    }
    if (symbol.equals("-")) {
        return "-(" + SLOT + ")";
    }
    if (symbol.equals("'")) {
        return SLOT + "'";
    }
    if (symbol.length() > 1 && symbol.startsWith(x)) {
        // postfix, like x!
        return SLOT + symbol.substring(x.length());
    }
    if (symbol.length() > 1 && symbol.endsWith(x)) {
        // prefix, like !x
        return symbol.substring(0, symbol.length() - x.length()) + SLOT;
    }
    return null;
}

public E eval(String input) {
    final String expression = input.strip();
    try (var r = ConfigurationService.setConfiguration(configuration)) {
        log.fine(() -> "Evaluating expression in %s: %s. Binary: %s, Unary: %s".formatted(field, expression, field.getSupportedOperators(), field.getSupportedUnaryOperators()));
        if (field.getSupportedOperators().isEmpty()) {
            log.log(Level.SEVERE,  "Supported operators is empty for " + field);
        }
        var parsedExpression = AST.parse(expression, field);
        log.fine(() -> "Parsed expression: %s".formatted(parsedExpression));
        return parsedExpression.eval();
    } catch (Throwable ex) {
        log.log(Level.SEVERE, ex.getClass() + " " + ex.getMessage(), ex);
        throw ex;
    } finally {
        log.finer("Ready evaluation");
    }
}

public String evalToString(String input) {
    var result = eval(input);
    var resultAsString = result.toString();
    log.fine(() -> "Result: %s = %s".formatted(input.strip(), resultAsString));
    return resultAsString;
}

/**
 * Evaluates, but never throws. Meant for the web app.
 * @param result  the result as a string, or {@code null} if evaluation failed
 * @param operand the result in a form that can be used as an operand in a next expression (see {@link AST#asOperand(String)}), or {@code null}
 * @param error   a message describing why evaluation failed, or {@code null}
 */
public record Result(String input, String result, String operand, String error) {
}

public Result calculate(String input) {
    try {
        String result = evalToString(input);
        return new Result(input, result, AST.asOperand(result), null);
    } catch (RuntimeException e) {
        String message = e.getMessage() == null || e.getMessage().isBlank() ? e.getClass().getSimpleName() : e.getMessage().strip();
        return new Result(input, null, null, message);
    }
}

    constructor() {
        super('#calculator', 'org.meeuw.math.demo.Calculator');
        // the keypad buttons are in the form too, so be explicit about which one submits
        this.button = this.form.querySelector('button[name=submit]');

        this.inputDataList= this.form.querySelector('datalist');
        this.tape = this.form.querySelector('ol.tape');
        this.help = this.form.querySelector('span.help');
        this.keypad = document.querySelector('#calculator_keypad');
        this.information = null;
        this.calculators = {};
        this.listenersSetUp = false;
        // true when the input contains the result of the previous calculation (completely selected)
        this.fresh = false;
        this.previewTimer = null;
        this.previewSequence = 0;
    }

    get fieldInformation() {
        return this.information[this.field.value];
    }

    /**
     * One Calculator instance per field
     */
    async calculator() {
        const name = this.field.value;
        if (!this.calculators[name]) {
            const Calculator = this.Class;
            this.calculators[name] = await new Calculator(this.fieldInformation.value);
        }
        return this.calculators[name];
    }

    /**
     * Inserts the string at the cursor, replacing the selection (if any)
     */
    insert(string, cursorOffset = string.length) {
        const input = this.input;
        const start = input.selectionStart;
        const end = input.selectionEnd;
        const value = input.value;
        input.value = value.slice(0, start) + string + value.slice(end);
        input.setSelectionRange(start + cursorOffset, start + cursorOffset);
        this.fresh = false;
        input.focus();
        this.schedulePreview();
    }

    /**
     * Inserts a binary operator. If the input contains a previous result, then the calculation continues with that.
     */
    insertBinaryOperator(operator) {
        const input = this.input;
        if (this.fresh) {
            input.setSelectionRange(input.value.length, input.value.length);
        }
        const before = input.value.slice(0, input.selectionStart);
        const padded = (before === '' || before.endsWith(' ') ? '' : ' ') + operator + ' ';
        this.insert(padded);
    }

    /**
     * Applies a unary operator, given as a template like 'sqr(□)', '□!' or '-(□)'. If something is selected (e.g. the previous
     * result), that becomes the operand. Otherwise the template is inserted with the cursor where the operand should go.
     */
    insertUnaryOperator(template) {
        const input = this.input;
        const [before, after] = template.split(SLOT);
        const selected = input.value.slice(input.selectionStart, input.selectionEnd).trim();
        if (selected !== '') {
            const bracketed = template.includes('(' + SLOT + ')');
            const operand = bracketed || SIMPLE_OPERAND.test(selected) ? selected : '(' + selected + ')';
            this.insert(before + operand + after);
        } else {
            // postfix operators (like □!) go after the cursor, others leave the cursor at the operand's place
            this.insert(before + after, before === '' ? after.length : before.length);
        }
    }

    backspace() {
        const input = this.input;
        let start = input.selectionStart;
        const end = input.selectionEnd;
        if (start === end && start > 0) {
            start--;
        }
        input.value = input.value.slice(0, start) + input.value.slice(end);
        input.setSelectionRange(start, start);
        this.fresh = false;
        input.focus();
        this.schedulePreview();
    }

    /**
     * Replaces the complete input (e.g. with an expression from the tape)
     */
    setInput(value) {
        this.input.value = value;
        this.input.setSelectionRange(value.length, value.length);
        this.fresh = false;
        this.input.focus();
        this.schedulePreview();
    }

    clear() {
        this.input.value = '';
        this.clearOutput();
        this.fresh = false;
        this.input.focus();
    }

    clearOutput() {
        this.output.value = '';
        this.output.classList.remove('error', 'preview');
    }

    /**
     * Whether typing '*' should produce '⋅'. Only if this field has no '*' operator of its own (in groups '*' is the group operation)
     */
    remapStar() {
        const operators = this.fieldInformation.binaryOperators;
        return operators.includes('⋅') && !operators.includes('*');
    }

    async setupForm() {
        await super.setupForm();
        if (this.listenersSetUp) {
            // setupForm is called every time the form comes into view
            return;
        }
        this.listenersSetUp = true;
        this.input.addEventListener('beforeinput', async (e) => {
            this.help.textContent = '';
            if (!e.inputType.startsWith('insert') || !e.data || e.data.length !== 1) {
                // deletions, pastes, etc.
                this.fresh = false;
                return;
            }
            let data = e.data;
            if (data === '=') {
                e.preventDefault();
                await this.handleSubmit();
                return;
            }
            if (data === ';') {
                // shortcut for '*', which can be typed without shift
                data = this.fieldInformation.binaryOperators.includes('*') ? '*' : '⋅';
            } else if (data === '*' && this.remapStar()) {
                data = '⋅';
                this.help.textContent = "(* is ⋅ here)";
            }
            if (this.fresh && this.fieldInformation.binaryOperators.includes(data)) {
                // continue with the previous result
                e.preventDefault();
                this.insertBinaryOperator(data);
            } else if (data !== e.data) {
                e.preventDefault();
                this.insert(data);
            } else {
                // just typing, which replaces the previous result if that was there
                this.fresh = false;
            }
        });
        this.input.addEventListener('input', () => {
            this.schedulePreview();
        });
        this.input.addEventListener('keydown', (e) => {
            if (e.key === 'Escape') {
                e.preventDefault();
                this.clear();
            }
        });
        this.field.addEventListener('change', async () => {
            await this.updateField();
        });
    }

    keyButton(label, title, onclick, ...classes) {
        const button = document.createElement('button');

    /**
     * (Re)builds the on-screen keypad for the current field
     */
    async updateKeypad() {
        if (!this.keypad) {
            return;
        }
        const name = this.field.value;
        const keys = await this.loadKeys(this.fieldInformation);
        if (name !== this.field.value) {
            // field changed again in the meantime
            return;
        }
        const groups = {
            element: this.keypad.querySelector('.elements'),
            binary: this.keypad.querySelector('.binary'),
            unary: this.keypad.querySelector('.unary'),
        };
        Object.values(groups).forEach(g => g.innerHTML = '');
        for (const key of keys) {
            let onclick;
            switch (key.type) {
                case 'binary':
                    onclick = () => this.insertBinaryOperator(key.insert);
                    break;
                case 'unary':
                    onclick = () => this.insertUnaryOperator(key.insert);
                    break;
                default:
                    onclick = () => this.insert(key.insert);
            }
            groups[key.type]?.appendChild(this.keyButton(key.label, key.title, onclick, 'key', key.type));
        }
        const special = this.keypad.querySelector('.special');
        if (special.childElementCount === 0) {
            special.append(
                this.keyButton('(', 'opening bracket', () => this.insert('(')),
                this.keyButton(')', 'closing bracket', () => this.insert(')')),
                this.keyButton('" "', 'quotes, for an element in its own notation, like "1 + 2i"', () => this.insert('""', 1)),
                this.keyButton('⌫', 'backspace', () => this.backspace()),
                this.keyButton('C', 'clear (Esc)', () => this.clear()),
            );
            special.querySelectorAll('button').forEach(b => b.classList.add('key', 'special'));
        }
    }

    /**
     * Evaluates while typing (a bit delayed), and shows the result as a preview
     */
    schedulePreview() {
        clearTimeout(this.previewTimer);
        this.previewTimer = setTimeout(() => this.preview(), 400);
    }

    async preview() {
        const expression = this.input.value.trim();
        const sequence = ++this.previewSequence;
        if (this.running || this.fresh || !this.Class || this.Class === "locked") {
            return;
        }
        if (expression === '') {
            this.clearOutput();
            return;
        }
        const result = await (await this.calculator()).calculate(expression);
        if (sequence !== this.previewSequence || this.running || this.input.value.trim() !== expression) {
            // outdated
            return;
        }
        const error = await result.error();
        this.output.classList.remove('error');
        if (error) {
            // probably just not finished typing
            this.output.classList.remove('preview');
            this.output.value = '';
        } else {
            this.output.classList.add('preview');
            this.output.value = '= ' + await result.result();
        }
    }

    addToTape(expression, result, operand) {
        const li = document.createElement('li');
        const expressionSpan = document.createElement('span');
        expressionSpan.classList.add('expression');
        expressionSpan.textContent = expression;
        expressionSpan.title = 'use this expression again';
        expressionSpan.onclick = () => this.setInput(expression);
        const resultSpan = document.createElement('span');
        resultSpan.classList.add('result');
        resultSpan.textContent = result;
        resultSpan.title = 'use this result';
        resultSpan.onclick = () => this.setInput(operand);
        li.append(expressionSpan, ' = ', resultSpan);
        this.tape.appendChild(li);
        this.tape.scrollTop = this.tape.scrollHeight;
    }

    async onSubmit() {
        clearTimeout(this.previewTimer);
        this.previewSequence++;
        const expression = this.input.value.trim();
        this.clearOutput();
        if (expression === '') {
            return;
        }
        const result = await (await this.calculator()).calculate(expression);
        const error = await result.error();
        if (error) {
            // leave the input, so it can be corrected
            this.output.classList.add('error');
            this.output.value = error;
            this.input.focus();
            return;
        }
        const value = await result.result();
        const operand = await result.operand();
        this.output.value = value;
        this.addToTape(expression, value, operand);
        // the result becomes the start of the next calculation. It is selected, so typing something new replaces it,
        // and typing an operator continues with it
        this.input.value = operand;
        this.input.focus();
        this.input.select();
        this.fresh = true;
    }
}

Solver

The same idea as my very first applet. Solving the '24 flippo game'.

Give the desired outcome number, and a few input numbers, and it will find the possible ways to get it using those input numbers.

Note

This combines several aspects of this project:

  • It uses 'rational numbers' to make all operations exact

  • It uses the permutation group to permute all combinations of values

  • It uses the Abstract Syntax Tree feature to combine values and operators

  • It uses some features of the Field, e.g. to perform operators generically.

  • It will currently use the field of RationalNumbers or GaussianRationals if the input contains a complex number. I might add also support for RealField (and for example also support the POW operator)
Note

This runs webassembly, so it is considerably slower than a pure java version. It is still fast enough for demo purposes. But e.g. the 175000 considerations to be made for 5 numbers '4 7 7 7 8' will cost several minutes in the browser. It runs in under 2 seconds in pure java.

import lombok.extern.java.Log;

import java.time.Duration;
import java.util.*;
import java.util.concurrent.CancellationException;
import java.util.concurrent.atomic.AtomicLong;
import java.util.function.Function;
import java.util.logging.Level;
import java.util.stream.Collectors;
import java.util.stream.Stream;

import org.meeuw.math.abstractalgebra.Ring;
import org.meeuw.math.abstractalgebra.RingElement;
import org.meeuw.math.abstractalgebra.complex.GaussianRationals;
import org.meeuw.math.abstractalgebra.permutations.PermutationGroup;
import org.meeuw.math.abstractalgebra.quaternions.Quaternions;
import org.meeuw.math.abstractalgebra.rationalnumbers.RationalNumbers;
import org.meeuw.math.arithmetic.ast.*;
import org.meeuw.math.exceptions.MathException;
import org.meeuw.math.exceptions.NotParsable;
import org.meeuw.math.operators.AlgebraicBinaryOperator;

import static org.meeuw.math.CollectionUtils.navigableSet;
import static org.meeuw.math.operators.BasicAlgebraicBinaryOperator.*;

/**
 * A tool to evaluate all possible expressions (of a certain number of rational numbers) (and check if it equals a certain value)
 */
@Log
public  class Solver<E extends RingElement<E>> {
    private static final NavigableSet<AlgebraicBinaryOperator> OPERATORS = navigableSet(
        ADDITION, SUBTRACTION, MULTIPLICATION, DIVISION
    );

    private final AtomicLong tries = new AtomicLong();
    private final AtomicLong considered = new AtomicLong();

    @Getter
    private final Ring<E> structure;

    public Solver(Ring<E> structure) {
        this.structure = structure;
    }

    native void callBack(long considered, long tried, long total, Expression<E> expression);

    native boolean cancelled();

    @SafeVarargs
    public final Stream<Expression<E>> stream(E... set) {
        PermutationGroup permutations = PermutationGroup.ofDegree(set.length);
        long streamSize = permutations.getCardinality().getValue().longValue() * AST.streamSize(set.length, OPERATORS.size()).longValue();

        long dividor = Arrays.stream(set)
            .collect(Collectors.groupingBy(
                Function.identity(),
                Collectors.counting()
            )).values().stream()
             .map(count -> IntegerUtils.factorial(count.intValue()))
            .reduce(1L, (a, b) -> a * b);
        long total = streamSize / dividor;

        return permutations.stream()
            .map(permutation -> permutation.permute(set))
            .map(List::of)
            .distinct()
            .flatMap(permuted ->
                AST.stream(
                    permuted,
                    OPERATORS
                )
            )
            .peek((p) -> {
                if (cancelled()) {
                    throw new CancellationException("cancelled");
                }
            })
            .map( e -> e.canonize(structure))
            .peek(e -> {
                considered.incrementAndGet();
            })
            .distinct()
            .peek(e ->
                callBack(considered.get(), tries.incrementAndGet(), total, e)
            );
    }



    @SafeVarargs
    public final Stream<EvaluatedExpression<E>> evaledStream(E... set) {
        log.fine(() -> "evaling" + List.of(set));
        return stream(set)
            .map(e -> {
                try {
                    log.fine(() -> "Evaling " +  e);
                    E evaled = e.eval();
                    return new EvaluatedExpression<>(e, evaled);
                } catch (MathException ex) {
                    return null;
                }
            })
            .filter(Objects::nonNull);
    }

    /**
     *
     */
    public SolverResult solve(String outcomeString, String inputStrings) {

        ParseResult<E> outcome = parseOutcome(outcomeString);
        ParseResult<E[]> input = parseInput(inputStrings);
        if (outcome.success() && input.success()) {
            SolverResult result = solve(outcome.result(), input.result());
            log.fine(() -> "solved: " + result);
            return result;
        } else {
            throw new NotParsable(outcome.error() + "/" + input.error());
        }
    }


    public SolverResult solve(E outcome, E[] input) {
        AtomicLong matches = new AtomicLong();
        log.fine(() -> "Solving input " + List.of(input) + " for " + outcome + " ( in field " + structure + ")");

        return new SolverResult(
            evaledStream(input)
            .filter(e ->
                e.result().eq(outcome)
            ).peek(e -> matches.getAndIncrement())
                .map(EvaluatedExpression::toString),
            tries, matches, structure
        );
    }


    public static <F extends RingElement<F>> ParseResult<F> parseOutcome(Ring<F> structure, String outcomeString) {
        log.fine(() -> "Parsing output " + outcomeString + " in field " + structure);

        String resultError = null;
        F result;
        try {
            result = structure.fromString(outcomeString);
        } catch (NotParsable pe) {
            result = null;
            resultError = pe.getMessage();
        }
        return new ParseResult<>(outcomeString, result, resultError);
    }
    public ParseResult<E> parseOutcome(String outcomeString) {
        return parseOutcome(this.structure, outcomeString);
    }

    public static <F extends RingElement<F>> ParseResult<F[]> parseInput(Ring<F> structure, String inputStrings) {
        log.fine(() -> "Parsing input " + inputStrings + " in field " + structure);

        String inputError = null;

        String[] input = inputStrings.split("\\s+");
        F[] set = structure.newArray(input.length);
        try {
            for (int i = 0; i < set.length; i++) {
                set[i] = structure.fromString(input[i]);
            }
        } catch (NotParsable pe) {
            inputError = pe.getMessage();
        }
        return new ParseResult<>(inputStrings, set, inputError);
    }
    public ParseResult<E[]> parseInput(String inputStrings) {
        return parseInput(this.structure, inputStrings);
    }


    public static Ring<?> algebraicStructureFor(String outcomeString, String input) {
        log.info(() -> "Determining algebraic structure for outcome " + outcomeString + " and input " + input);
        if (outcomeString.matches(".*[jk].*") || input.matches(".*[jk].*")) {
            return Quaternions.of(RationalNumbers.INSTANCE);
        } else if (outcomeString.contains("i") || input.contains("i")) {
            return GaussianRationals.INSTANCE;
        } else {
            return RationalNumbers.INSTANCE;
        }
    }

    public record SolverResult(Stream<String> stream, AtomicLong tries, AtomicLong matches, Ring<?> field) {
        /**
         * Calling stream().iterator() in cheerpj doesn't seem to work. I don't get it either.
         */
        public Iterator<String> iterator() {
            return stream().iterator();
        }

        public List<String> list() {
            return stream().toList();
        }
    }

    /**
     * The natives are meant to implement by CheerpJ
     * This is a dummy implementation for use in main.
     * @param <E>
     */
    static class Impl<E extends RingElement<E>> extends Solver<E> {

        private final Level level;
        public Impl(Ring<E> structure, Level level) {
            super(structure);
            this.level = level;
        }

        @Override
        void callBack(long considered, long tried, long total, Expression<E> expression) {
            if (considered % 10000 == 0 || (considered < 10000 && considered % 100 == 0)) {
                log.log(level,"Considered %d / %d , tried %d : %s".formatted(considered, total, tried, expression));
            }
        }

        @Override
        boolean cancelled() {
            return false;
        }
    }

    static void main(String[] integers) {
        if (integers.length < 3) {
            System.out.println();
            System.exit(1);
        }
        String resultString = integers[0];
        String inputStrings = String.join(" ", Arrays.copyOfRange(integers, 1, integers.length));

        Ring<?> field = algebraicStructureFor(resultString, inputStrings);

        long start = System.nanoTime();
        Solver<?> solver = new Impl<>(field, Level.FINE);
        SolverResult solverResult = solver.solve(resultString, inputStrings);
        solverResult.stream().forEach(System.out::println);
        System.out.println("ready, found " + solverResult.matches().get() + ", tried " + solverResult.tries.get() + ", field " + solverResult.field().toString() + ", took " + Duration.ofNanos(System.nanoTime() - start));
    }
}
